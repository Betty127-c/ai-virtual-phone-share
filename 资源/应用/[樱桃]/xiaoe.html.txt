/* ================= 工具 ================= */
function money(v){
  if (v >= 10000){
    const w = Math.round(v / 1000) / 10;
    return '¥' + w + '万';
  }
  return '¥' + (Number.isInteger(v) ? v : v.toFixed(2));
}
function calc(){
  let count = 0, total = 0;
  for (const id in cart){
    count += cart[id];
    total += itemMap[id].price * cart[id];
  }
  return { count, total };
}
let toastTimer;
function toast(msg, ms){
  let el = document.querySelector('.toast');
  if (!el){ el = document.createElement('div'); el.className = 'toast'; document.body.appendChild(el); }
  el.textContent = msg;
  clearTimeout(toastTimer);
  requestAnimationFrame(() => el.classList.add('on'));
  toastTimer = setTimeout(() => el.classList.remove('on'), ms || 1800);
}

/* ================= 商品渲染 ================= */
function renderGrid(){
  const q = searchEl.value.trim();
  let list, emptyTip = '';

  if (q){
    list = Object.values(itemMap).filter(o =>
      o.name.includes(q) || o.desc.includes(q) || (o.tag && o.tag.includes(q))
    );
    emptyTip = `没有找到「${q}」相关商品`;
  } else {
    const c = CATS.find(c => c.k === curCat);
    list = c.list;
    emptyTip = c.k === 'ai'
      ? '这里还空空的<br>点右上角 ✨ 让 AI 为你生成商品'
      : '暂无商品';
  }

  if (!list.length){
    gridEl.innerHTML = `<div class="empty">${emptyTip}</div>`;
    return;
  }

  gridEl.innerHTML = list.map(o => {
    const q2 = cart[o.id] || 0;
    return `<div class="card${q2 ? ' has' : ''}" data-id="${o.id}">
      <span class="cbadge">${q2 || ''}</span>
      <div class="ico">${o.emoji}</div>
      <div class="nm">${o.name}</div>
      <div class="pr">${money(o.price)}</div>
      <button class="plus">+</button>
    </div>`;
  }).join('');
}

gridEl.addEventListener('click', e => {
  const card = e.target.closest('.card');
  if (!card) return;
  addToCart(card.dataset.id);
});

function addToCart(id){
  cart[id] = (cart[id] || 0) + 1;
  syncCards();
  syncBar();
}

function syncCards(){
  document.querySelectorAll('.card').forEach(el => {
    const q = cart[el.dataset.id] || 0;
    el.classList.toggle('has', q > 0);
    el.querySelector('.cbadge').textContent = q || '';
  });
}

function syncBar(){
  const { count, total } = calc();
  badgeEl.textContent = count;
  badgeEl.style.display = count ? 'grid' : 'none';
  totalEl.textContent = money(total);

  if (count && total >= MIN_ORDER){
    coBtn.classList.add('on');
    coBtn.textContent = '结算';
  } else {
    coBtn.classList.remove('on');
    coBtn.textContent = count ? `差 ${money(MIN_ORDER - total)} 起送` : `¥${MIN_ORDER} 起送`;
  }
  renderCart();
}

      <div class="cinfo">

}

});

  document.body.removeChild(ta);

怎么导入小手机

方法一：浏览器直接打开（最简单）

1. 将上面的完整代码复制
2. 在电脑上新建一个文本文件，粘贴代码，保存为 xiaoe.html（注意后缀是 .html 不是 .txt）
3. 通过微信/QQ/数据线/网盘把文件传到手机
4. 在手机的文件管理器里找到 xiaoe.html，点击选择用浏览器打开
5. 打开后，浏览器菜单里选“添加到主屏幕”，就会生成一个带图标的小应用

方法二：手机端直接创建

1. 在手机上下载“HTML Viewer”或“代码编辑器”类App
2. 新建文件，粘贴代码，保存为 .html
3. 用App内置浏览器打开即可

小提示：添加到主屏幕后，图标会显示红色购物袋，点开就是全屏应用，没有浏览器地址栏，体验接近原生App。
