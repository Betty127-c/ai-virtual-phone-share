  // 要求：歌名 -> 作词 -> 作曲 -> 真歌词

  const lyricsContainer = document.getElementById('lyrics-container');

  lyricsData.forEach((text) => {
    const div = document.createElement('div');
    div.className = 'lyric-line';
    div.textContent = text;
    lyricsContainer.appendChild(div);
  });

  const lines = document.querySelectorAll('.lyric-line');
  const lineHeight = 38; 
  const totalDuration = 320; // 5分20秒
  const lineDuration = totalDuration / lyricsData.length;

  let currentTime = 0; 
  let currentIndex = 3; // 从第一句真歌词开始
  let isPlaying = true;
  let playTimer = null;

  // 初始化滚动位置和遮罩
  function initScroll() {
    // 给上下加上 padding，保证第一句能居中
    const containerHeight = lyricsContainer.clientHeight;
    const padding = (containerHeight - lineHeight) / 2;
    lyricsContainer.style.paddingTop = `${padding}px`;
    lyricsContainer.style.paddingBottom = `${padding}px`;

    // 瞬间滚动到第一句
    lyricsContainer.style.scrollBehavior = 'auto';
    lyricsContainer.scrollTop = currentIndex * lineHeight;
    
    setTimeout(() => {
      lyricsContainer.style.scrollBehavior = 'smooth';
    }, 50);
    
    updateVisuals();
  }

  function formatTime(sec) {
    let m = Math.floor(sec / 60);
    let s = Math.floor(sec % 60);
    return `0${m}:${s < 10 ? '0'+s : s}`;
  }

  function updateVisuals() {
    lines.forEach((line, idx) => {
      line.className = 'lyric-line'; 
      const diff = idx - currentIndex;
      if (diff === 0) {
        line.classList.add('alpha');
      } else if (Math.abs(diff) === 1) {
        line.classList.add('a-level');
      } else if (Math.abs(diff) === 2) {
        line.classList.add('b-level');
      }
    });
  }

  function tick() {
    if (!isPlaying) return;
    currentTime += 0.1; 

    if (currentTime >= totalDuration) {
      currentTime = 0;
      currentIndex = 3;
      
      lyricsContainer.style.scrollBehavior = 'auto';
      lyricsContainer.scrollTop = currentIndex * lineHeight;
      updateVisuals();
      
      setTimeout(() => {
        lyricsContainer.style.scrollBehavior = 'smooth';
      }, 50);
      
      progressFill.style.width = '0%';
      progressDot.style.left = '0%';
      timeDisplay.textContent = `00:00 / 05:20`;
      return;
    }

    const percent = (currentTime / totalDuration) * 100;
    progressFill.style.width = `${percent}%`;
    progressDot.style.left = `${percent}%`;
    timeDisplay.textContent = `${formatTime(currentTime)} / 05:20`;

    const targetIndex = Math.floor(currentTime / lineDuration) + 3;
    
    if (targetIndex !== currentIndex && targetIndex < lyricsData.length) {
      currentIndex = targetIndex;
      // 使用原生滚动
      const targetScroll = currentIndex * lineHeight;
      lyricsContainer.scrollTo({
        top: targetScroll,
        behavior: 'smooth'
      });
      updateVisuals(); 
    }
  }

  function startPlaying() {
    if (playTimer) clearInterval(playTimer);
    playTimer = setInterval(tick, 100);
  }

  playBtn.addEventListener('click', () => {
    isPlaying = !isPlaying;
    if (isPlaying) {
      iconPlay.style.display = 'none';
      iconPause.style.display = 'block';
    } else {
      iconPlay.style.display = 'block';
      iconPause.style.display = 'none';
    }
  });

  setTimeout(() => {
    initScroll();
    timeDisplay.textContent = `00:00 / 05:20`;
    startPlaying();

  const whiteKeys = Array.from(document.querySelectorAll('.key-white'));
  const blackKeys = Array.from(document.querySelectorAll('.key-black'));
  const wordsContainer = document.getElementById('words-container');
  const sandboxContainer = document.querySelector('.sandbox-container');

  const words = [
    'Love', 'Star', 'Dream', 'Echo', 'Aura', 'Muse', 'Whisper', 
    'Memory', 'Eternity', 'Apricity', 'Petrichor', 'Nefelibata'
  ];
  const symbols = ['𝆹𝅥𝅮', '𝆺𝅥𝅮', '𝆹𝅥𝅯', '𝆺𝅥𝅯', '𝅗𝅥', '𝅘𝅥', '𝅘𝅥𝅮', '𝅘𝅥𝅯', '𝅘𝅥𝅰', '𝅘𝅥𝅱', '𝅘𝅥𝅲', '𝄞', '𝄟', '𝄠', '♩', '♪', '♫', '♬', '♭', '♮'];
  const mixedPool = [...words, ...symbols];

  const allRealKeys = [...whiteKeys, ...blackKeys];
  let playQueue = [];

  function shuffleArray(array) {
    const arr = [...array];
    for (let i = arr.length - 1; i > 0; i--) {
      const j = Math.floor(Math.random() * (i + 1));
      [arr[i], arr[j]] = [arr[j], arr[i]];
    }
    return arr;
  }

  function playNextNote() {
    if (playQueue.length < 3) {
      playQueue = playQueue.concat(shuffleArray(allRealKeys));
    }

    // 1. 将和弦概率提升到 55%，让画面更满
    let isChord = Math.random() < 0.55; 
    
    const keyA = playQueue.pop();
    const idxA = allRealKeys.indexOf(keyA);
    let keyB = null;
    let idxB = -1;

    if (isChord) {
      for (let i = playQueue.length - 1; i >= 0; i--) {
        const potentialKeyB = playQueue[i];
        const potentialIdxB = allRealKeys.indexOf(potentialKeyB);
        
        if (Math.abs(idxA - potentialIdxB) >= 2) {
          keyB = potentialKeyB;
          idxB = potentialIdxB;
          playQueue.splice(i, 1);
          break;
        }
      }
      if (!keyB) {
        playQueue.push(keyA);
        playQueue = shuffleArray(playQueue);
        return playNextNote(); 
      }
    }

    if (keyB) {
      // 2. 和弦：一个英文 + 一个符号
      const randomWord = words[Math.floor(Math.random() * words.length)];
      const randomSymbol = symbols[Math.floor(Math.random() * symbols.length)];
      
      if (idxA < idxB) {
        executeNote(keyA, randomWord, true); 
        setTimeout(() => executeNote(keyB, randomSymbol, true), 120); 
      } else {
        executeNote(keyA, randomWord, true);
        setTimeout(() => executeNote(keyB, randomSymbol, true), 120);
      }
    } else {
      // 3. 单键：从混合池子里随机抽，并给一个随机的左右偏移量（-20 到 20）
      const randomItem = mixedPool[Math.floor(Math.random() * mixedPool.length)];
      const randomOffset = (Math.random() - 0.5) * 40; // 单键也不再死板在正中间
      executeNote(keyA, randomItem, false, randomOffset);
    }

    // 4. 节奏加快，不再空旷
    let nextTime;
    const rhythmType = Math.random();
    if (rhythmType < 0.15) {
      nextTime = 1200 + Math.random() * 200; 
    } else if (rhythmType < 0.6) {
      nextTime = 800 + Math.random() * 200; 
    } else {
      nextTime = 450 + Math.random() * 200;  
    }

    setTimeout(playNextNote, nextTime);
  }

  function executeNote(key, textToDisplay = '', isChord = false, singleKeyOffset = 0) {
    key.classList.add('active');
    setTimeout(() => key.classList.remove('active'), 150);

    const rect = key.getBoundingClientRect();
    const containerRect = sandboxContainer.getBoundingClientRect();
    const keyCenterX = rect.left - containerRect.left + (rect.width / 2);
    const startY = rect.top - containerRect.top - 15; 

    createFloatingWord(keyCenterX, startY, textToDisplay, isChord, singleKeyOffset);
  }

  function createFloatingWord(x, y, text, isChord, singleKeyOffset) {
    const wordEl = document.createElement('div');
    wordEl.className = 'floating-word';
    wordEl.textContent = text;

    // 随机大小
    const baseSize = 16;
    const randomScale = 0.7 + Math.random() * 0.8; 
    wordEl.style.fontSize = `${baseSize * randomScale}px`;

    // ==========================================
    // 核心防截断算法
    // ==========================================
    // 如果是和弦，给一个随机的偏移量，如果单键，用传进来的偏移量
    let offsetX = isChord ? (Math.random() > 0.5 ? 1 : -1) * (Math.random() * 30 + 20) : singleKeyOffset;
    
    // 预估文字的宽度 (粗略估算，长词会更宽)
    const textWidth = text.length * (baseSize * randomScale) * 0.6;
    
    // 容器（屏幕）的宽度
    const canvasWidth = sandboxContainer.clientWidth;
    
    // 计算最终 X 坐标
    let finalX = x + offsetX;
    
    // 如果文字太长，快要超出左边界了，就强行往右拉
    if (finalX - textWidth / 2 < 10) {
      finalX = textWidth / 2 + 10;
    }
    // 如果快要超出右边界了，就强行往左拉
    else if (finalX + textWidth / 2 > canvasWidth - 10) {
      finalX = canvasWidth - textWidth / 2 - 10;
    }

    wordEl.style.left = `${finalX}px`;
    wordEl.style.top = `${y}px`;

    wordsContainer.appendChild(wordEl);

    setTimeout(() => wordEl.remove(), 2500);
  }

  setTimeout(playNextNote, 500);

  /* ================= 1. 星空系统 ================= */
  const canvasBack = document.getElementById('star-canvas-back');
  const canvasFront = document.getElementById('star-canvas-front');
  const ctxBack = canvasBack.getContext('2d');
  const ctxFront = canvasFront.getContext('2d');

  let width = window.innerWidth;
  let height = window.innerHeight;
  canvasBack.width = width; canvasBack.height = height;
  canvasFront.width = width; canvasFront.height = height;

  window.addEventListener('resize', () => {
    width = window.innerWidth; height = window.innerHeight;
    canvasBack.width = width; canvasBack.height = height;
    canvasFront.width = width; canvasFront.height = height;
  });

  function drawStarShape(ctx, x, y, size, alpha) {
    ctx.save();
    ctx.translate(x, y);
    
    ctx.beginPath();
    ctx.arc(0, 0, size * 0.7, 0, Math.PI * 2);
    ctx.fillStyle = `rgba(255, 255, 255, ${alpha * 0.6})`; 
    ctx.shadowBlur = size * 2; 
    ctx.shadowColor = `rgba(255, 255, 255, ${alpha * 0.4})`;
    ctx.fill();

    if (size > 1.0) {
      ctx.beginPath();
      const armLength = size * 3;
      ctx.moveTo(-armLength, 0); ctx.lineTo(armLength, 0);
      ctx.moveTo(0, -armLength); ctx.lineTo(0, armLength);
      ctx.strokeStyle = `rgba(255, 255, 255, ${alpha * 0.35})`; 
      ctx.lineWidth = Math.max(0.3, size * 0.15);
      ctx.stroke();
    }
    ctx.restore();
  }

  class TwinkleStar {
    constructor() { this.reset(); this.y = Math.random() * height; }
    reset() {
      this.x = Math.random() * width; this.y = Math.random() * height;
      this.size = Math.random() * 1.5 + 0.3; 
      this.alpha = Math.random();
      this.twinkleSpeed = Math.random() * 0.03 + 0.01;
    }
    update() {
      this.alpha += this.twinkleSpeed;
      if (this.alpha > 1 || this.alpha < 0.1) this.twinkleSpeed *= -1;
    }
    draw(ctx) { drawStarShape(ctx, this.x, this.y, this.size, Math.abs(this.alpha)); }
  }

  class FloatingStar {
    constructor() { this.reset(); }
    reset() {
      this.x = Math.random() * width + 100; this.y = height + 20;
      this.size = Math.random() * 1.2 + 0.4; 
      this.alpha = Math.random();
      this.twinkleSpeed = Math.random() * 0.02 + 0.01;
      this.speedX = -(Math.random() * 0.3 + 0.1); this.speedY = -(Math.random() * 0.5 + 0.2); 
    }
    update() {
      this.x += this.speedX; this.y += this.speedY;
      this.alpha += this.twinkleSpeed;
      if (this.alpha > 1 || this.alpha < 0.2) this.twinkleSpeed *= -1;
      if (this.y < -20 || this.x < -20) this.reset();
    }
    draw(ctx) { drawStarShape(ctx, this.x, this.y, this.size, Math.abs(this.alpha)); }
  }

  class Meteor {
    constructor() {
      this.isAlive = false;
      this.timer = Math.random() * 3000; 
    }
    reset() {
      if (Math.random() > 0.5) {
        this.x = width + 50; this.y = Math.random() * height; 
      } else {
        this.x = Math.random() * width; this.y = -50;
      }
      this.speedX = -(Math.random() * 7 + 4); 
      this.speedY = Math.abs(this.speedX) * (0.7 + Math.random() * 0.3);
      
      this.length = Math.random() * 45 + 30; 
      this.headRadius = Math.random() * 2.0 + 1.0; 
      this.isAlive = true;
    }
    update() {
      if (!this.isAlive) {
        this.timer -= 16.6; 
        if (this.timer <= 0) { this.reset(); }
        return;
      }
      this.x += this.speedX; this.y += this.speedY;
      if (this.x < -100 || this.y > height + 100) {
        this.isAlive = false; 
        this.timer = Math.random() * 1500 + 1000; 
      }
    }
    draw(ctx) {
      if (!this.isAlive) return;
      const tailX = this.x - (this.length * (this.speedX / Math.abs(this.speedY)));
      const tailY = this.y - this.length;
      let gradient = ctx.createLinearGradient(this.x, this.y, tailX, tailY);
      
      gradient.addColorStop(0, `rgba(255, 255, 255, 0.9)`); 
      gradient.addColorStop(1, `rgba(255, 255, 255, 0)`);
      ctx.beginPath();
      ctx.moveTo(this.x, this.y); ctx.lineTo(tailX, tailY);
      ctx.strokeStyle = gradient;
      ctx.lineWidth = this.headRadius * 0.8; 
      ctx.lineCap = 'round'; ctx.stroke();

      ctx.beginPath();
      ctx.arc(this.x, this.y, this.headRadius, 0, Math.PI * 2);
      ctx.fillStyle = '#ffffff'; 
      ctx.shadowBlur = 15; 
      ctx.shadowColor = '#ffffff'; 
      ctx.fill(); ctx.shadowBlur = 0;
    }
  }

  const backTwinkles = Array.from({ length: 22 }, () => new TwinkleStar()); 
  const frontTwinkles = Array.from({ length: 13 }, () => new TwinkleStar()); 
  const backFloats = Array.from({ length: 8 }, () => new FloatingStar()); 
  const frontFloats = Array.from({ length: 5 }, () => new FloatingStar()); 
  const meteors = Array.from({ length: 18 }, () => new Meteor()); 

  function animateStars() {
    ctxBack.clearRect(0, 0, width, height);
    ctxFront.clearRect(0, 0, width, height);
    meteors.forEach(m => { m.update(); m.draw(ctxBack); });
    backTwinkles.forEach(t => { t.update(); t.draw(ctxBack); });
    backFloats.forEach(f => { f.update(); f.draw(ctxBack); });
    frontTwinkles.forEach(t => { t.update(); t.draw(ctxFront); });
    frontFloats.forEach(f => { f.update(); f.draw(ctxFront); });
    requestAnimationFrame(animateStars);
  }
  animateStars();


  /* ================= 2. 宿主环境图片读取与覆盖 ================= */
  function loadFromHostWidget() {
    if (window.AiPhoneWidget) {
      const cImg = window.AiPhoneWidget.getImage('center_img') || window.AiPhoneWidget.getConfig('center_img');
      if (cImg) { document.getElementById('img-center').src = cImg; }
      
      for (let i = 0; i < 5; i++) {
        const sImg = window.AiPhoneWidget.getImage('sat_img_' + i) || window.AiPhoneWidget.getConfig('sat_img_' + i);
        if (sImg) {
          const el = document.getElementById('img-sat-' + i);
          if (el) { el.src = sImg; }
        }
      }
    }
  }
  loadFromHostWidget();

  function setupHostInput(inputId, imgId, configKey) {
    const input = document.getElementById(inputId);
    const img = document.getElementById(imgId);
    if (!input || !img) return;
    input.addEventListener('change', function(e) {
      const file = e.target.files[0];
      if (!file) return;
      const reader = new FileReader();
      reader.onload = function(evt) {
        const dataUrl = evt.target.result;
        img.src = dataUrl; 
        if (window.AiPhoneWidget) {
          if (typeof window.AiPhoneWidget.setImage === 'function') window.AiPhoneWidget.setImage(configKey, dataUrl);
          if (typeof window.AiPhoneWidget.setConfig === 'function') window.AiPhoneWidget.setConfig(configKey, dataUrl);
        }
      };
      reader.readAsDataURL(file);
    });
  }

  setupHostInput('input-center', 'img-center', 'center_img');
  for (let i = 0; i < 5; i++) { setupHostInput('input-sat-' + i, 'img-sat-' + i, 'sat_img_' + i); }

  const RX = 125; 
  const RY = 36;  
  const SPEED = 0.0035; 

  let angle = 0;
  const count = 5;
  const step = (2 * Math.PI) / count;

  const satellites = [];
  for (let i = 0; i < count; i++) {
    satellites.push({ el: document.getElementById('sat-' + i), baseOffset: i * step });
  }

  const orbitNodes = [];
  for (let i = 0; i < count; i++) {
    orbitNodes.push({ el: document.getElementById('node-' + i), baseOffset: (i * step) + (step / 2) });
  }

  const cvsBack = document.getElementById('cvs-back');
  const cvsFront = document.getElementById('cvs-front');
  const ctxCometBack = cvsBack.getContext('2d');
  const ctxCometFront = cvsFront.getContext('2d');

  let cometAngle = 0;
  const COMET_SPEED = 0.042; 
  const TAIL_ANGLE = 1.5;    
  const TAIL_STEPS = 50;

  function renderComet() {
    ctxCometBack.clearRect(0, 0, 250, 72);
    ctxCometFront.clearRect(0, 0, 250, 72);
    cometAngle += COMET_SPEED;
    const cx = 125; const cy = 36;

    for (let i = TAIL_STEPS; i >= 1; i--) {
      const t0 = i / TAIL_STEPS;
      const t1 = (i - 1) / TAIL_STEPS;
      const a0 = cometAngle - (t0 * TAIL_ANGLE);
      const a1 = cometAngle - (t1 * TAIL_ANGLE);
      const x0 = cx + RX * Math.cos(a0);
      const y0 = cy + RY * Math.sin(a0);
      const x1 = cx + RX * Math.cos(a1);
      const y1 = cy + RY * Math.sin(a1);
      const midY = (y0 + y1) / 2;
      const targetCtx = (midY >= cy) ? ctxCometFront : ctxCometBack;
      const factor = 1 - t1;
      
      const alpha = Math.pow(factor, 1.2) * 0.95; 
      const width = 1.2 + factor * 3.5; 

      targetCtx.save();
      
      targetCtx.strokeStyle = `rgba(255, 255, 255, ${alpha * 0.3})`;
      targetCtx.lineWidth = width + 1.5; 
      targetCtx.lineCap = 'round';
      targetCtx.beginPath();
      targetCtx.moveTo(x0, y0);
      targetCtx.lineTo(x1, y1);
      targetCtx.stroke();

      targetCtx.strokeStyle = `rgba(255, 255, 255, ${alpha})`;
      targetCtx.lineWidth = width;
      targetCtx.shadowColor = '#ffffff';
      targetCtx.shadowBlur = 6 + factor * 12; 
      targetCtx.beginPath();
      targetCtx.moveTo(x0, y0);
      targetCtx.lineTo(x1, y1);
      targetCtx.stroke();
      
      targetCtx.restore();
    }

    const headX = cx + RX * Math.cos(cometAngle);
    const headY = cy + RY * Math.sin(cometAngle);
    const headCtx = (headY >= cy) ? ctxCometFront : ctxCometBack;

    headCtx.save();
    headCtx.shadowColor = '#ffffff';
    
    headCtx.shadowBlur = 15; 
    headCtx.fillStyle = '#ffffff';
    headCtx.beginPath();
    headCtx.arc(headX, headY, 3.0, 0, Math.PI * 2); 
    headCtx.fill();

    headCtx.shadowBlur = 5;
    headCtx.beginPath();
    headCtx.arc(headX, headY, 1.5, 0, Math.PI * 2); 
    headCtx.fill();
    headCtx.restore();
  }

  function renderOrbit() {
    angle += SPEED;
    satellites.forEach(sat => {
      const curAngle = angle + sat.baseOffset;
      const x = RX * Math.cos(curAngle);
      const y = RY * Math.sin(curAngle);
      const sinVal = Math.sin(curAngle);
      const scale = 0.92 + (sinVal * 0.25);
      
      sat.el.style.transform = `translate3d(${x}px, ${y}px, 0px) scale(${scale})`;
      sat.el.style.zIndex = sinVal > 0 ? 100 : 25;

      const satRing = sat.el.querySelector('.sat-ring');
      if (satRing) {
        const distX = Math.abs(x); 
        if (sinVal < 0) {
          if (distX < 55) {
            satRing.style.opacity = 0;
          } else if (distX < 75) {
            satRing.style.opacity = (distX - 55) / 20;
          } else {
            satRing.style.opacity = 1;
          }
        } else {
          satRing.style.opacity = 1; 
        }
      }
    });

    orbitNodes.forEach(node => {
      const curAngle = angle + node.baseOffset;
      const x = RX * Math.cos(curAngle);
      const y = RY * Math.sin(curAngle);
      const sinVal = Math.sin(curAngle);
      const scale = 0.9 + (sinVal * 0.22);
      const opacity = 0.75 + (sinVal * 0.25);
      node.el.style.transform = `translate3d(${x}px, ${y}px, 0px) scale(${scale})`;
      node.el.style.zIndex = sinVal > 0 ? 95 : 24;
      node.el.style.opacity = opacity;
    });

    renderComet();
    requestAnimationFrame(renderOrbit);
  }

  requestAnimationFrame(renderOrbit);
